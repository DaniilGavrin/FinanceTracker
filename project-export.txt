import { Debt } from '@/lib/database';

export function getPaymentDate(year: number, month: number, day: number): Date {
  const lastDay = new Date(year, month + 1, 0).getDate();
  const actualDay = Math.min(day, lastDay);
  return new Date(year, month, actualDay);
}

export interface PaymentRow {
  month: number;
  date: string;
  payment: number;
  interest: number;
  principal: number;
  remainingBalance: number;
}

export interface PaymentCalculation {
  monthlyPayment: number | null;
  totalPayments: number;
  totalInterest: number;
  totalPrincipal: number;
  schedule: PaymentRow[];
}

// Хелпер для округления до копеек
function roundToKopecks(value: number): number {
  return Math.round(value * 100) / 100;
}

// Хелпер для расчёта дней между датами (как в Сбере)
function calculateDaysBetween(date1: Date, date2: Date): number {
  const msPerDay = 1000 * 60 * 60 * 24;
  // Используем Math.round вместо Math.ceil
  return Math.round((date2.getTime() - date1.getTime()) / msPerDay);
}

export function calculatePaymentSchedule(debt: Debt): PaymentCalculation {
  if (
    debt.type !== 'bank_loan' ||
    !debt.termMonths ||
    !debt.interestRate ||
    !debt.startDate ||
    debt.totalAmount <= 0
  ) {
    return {
      monthlyPayment: null,
      totalPayments: debt.currentAmount,
      totalInterest: 0,
      totalPrincipal: debt.totalAmount,
      schedule: [],
    };
  }

  const isDiff = debt.paymentType === 'differentiated';
  const annualRate = debt.interestRate / 100;
  const months = debt.termMonths;
  const principal = debt.totalAmount;
  const startDate = new Date(debt.startDate);
  const paymentDay = debt.paymentDay || startDate.getDate();

  // Дата первого платежа
  let firstPaymentDate: Date;
  if (debt.nextPaymentDate) {
    firstPaymentDate = new Date(debt.nextPaymentDate);
    firstPaymentDate.setHours(0, 0, 0, 0);
    if (firstPaymentDate.getTime() < startDate.getTime()) {
      firstPaymentDate = new Date(startDate);
    }
  } else {
    firstPaymentDate = getPaymentDate(startDate.getFullYear(), startDate.getMonth(), 31);
    if (firstPaymentDate.getTime() <= startDate.getTime()) {
      const nextMonth = new Date(startDate);
      nextMonth.setMonth(nextMonth.getMonth() + 1);
      firstPaymentDate = getPaymentDate(nextMonth.getFullYear(), nextMonth.getMonth(), paymentDay);
    }
  }

  // Дата закрытия = startDate + months
  const closingDate = new Date(startDate);
  closingDate.setMonth(closingDate.getMonth() + months);

  const rows: PaymentRow[] = [];
  let balance = roundToKopecks(principal); // ✅ Округляем начальный баланс
  let totalInterest = 0;
  let prevDate = new Date(startDate);
  let monthNumber = 1;

  const hasPartialFirstPeriod = firstPaymentDate.getTime() > startDate.getTime();

  if (isDiff) {
    // ДИФФЕРЕНЦИРОВАННЫЙ
    const fixedPrincipal = roundToKopecks(principal / months);

    // Первый платёж (неполный период)
    if (hasPartialFirstPeriod) {
      const daysFirst = calculateDaysBetween(prevDate, firstPaymentDate); // ✅ Math.round
      const intFirst = roundToKopecks(balance * annualRate * daysFirst / 365); // ✅ Округляем проценты
      const payFirst = roundToKopecks(fixedPrincipal + intFirst);
      balance = roundToKopecks(balance - fixedPrincipal); // ✅ Округляем остаток
      totalInterest += intFirst;
      
      rows.push({
        month: monthNumber++,
        date: firstPaymentDate.toLocaleDateString('ru-RU'),
        payment: payFirst,
        interest: intFirst,
        principal: fixedPrincipal,
        remainingBalance: balance,
      });
      prevDate = firstPaymentDate;
    }

    // Полные платежи
    const fullPaymentsCount = hasPartialFirstPeriod ? months - 1 : months;
    for (let i = 1; i <= fullPaymentsCount; i++) {
      const base = new Date(firstPaymentDate);
      base.setMonth(base.getMonth() + i);
      const payDate = getPaymentDate(base.getFullYear(), base.getMonth(), paymentDay);
      if (payDate.getTime() >= closingDate.getTime()) continue;

      const days = calculateDaysBetween(prevDate, payDate); // ✅ Math.round
      const interest = roundToKopecks(balance * annualRate * days / 365); // ✅ Округляем
      
      let principalPart = fixedPrincipal;
      if (i === fullPaymentsCount) principalPart = balance; // Последний платёж — весь остаток
      
      const payment = roundToKopecks(principalPart + interest);
      balance = roundToKopecks(balance - principalPart); // ✅ Округляем остаток
      if (balance < 0) balance = 0;
      
      totalInterest += interest;
      
      rows.push({
        month: monthNumber++,
        date: payDate.toLocaleDateString('ru-RU'),
        payment,
        interest,
        principal: principalPart,
        remainingBalance: balance,
      });
      prevDate = payDate;
    }

    // Финальный платёж (если остался хвост)
    if (balance > 0.01) {
      const daysFinal = calculateDaysBetween(prevDate, closingDate); // ✅ Math.round
      const intFinal = roundToKopecks(balance * annualRate * daysFinal / 365);
      const payFinal = roundToKopecks(balance + intFinal);
      
      rows.push({
        month: monthNumber,
        date: closingDate.toLocaleDateString('ru-RU'),
        payment: payFinal,
        interest: intFinal,
        principal: balance,
        remainingBalance: 0,
      });
    }
  } else {
    // АННУИТЕТНЫЙ — ЛОГИКА СБЕРА
    const monthlyRate = annualRate / 12;
    const annuityPayment = roundToKopecks(
      principal * (monthlyRate * Math.pow(1 + monthlyRate, months)) /
      (Math.pow(1 + monthlyRate, months) - 1)
    ); // ✅ Округляем аннуитет сразу

    // Первый платёж — ТОЛЬКО ПРОЦЕНТЫ, БАЛАНС НЕ МЕНЯЕТСЯ
    if (hasPartialFirstPeriod) {
      const daysFirst = calculateDaysBetween(prevDate, firstPaymentDate); // ✅ Math.round
      const intFirst = roundToKopecks(balance * annualRate * daysFirst / 365); // ✅ Округляем
      // balance остаётся = principal (округлённый)
      totalInterest += intFirst;
      
      rows.push({
        month: monthNumber++,
        date: firstPaymentDate.toLocaleDateString('ru-RU'),
        payment: intFirst,
        interest: intFirst,
        principal: 0,
        remainingBalance: balance,
      });
      prevDate = firstPaymentDate;
    }

    // Полные аннуитетные платежи
    const fullPaymentsCount = hasPartialFirstPeriod ? months - 1 : months;
    for (let i = 1; i <= fullPaymentsCount; i++) {
      const base = new Date(firstPaymentDate);
      base.setMonth(base.getMonth() + i);
      const payDate = getPaymentDate(base.getFullYear(), base.getMonth(), paymentDay);
      if (payDate.getTime() >= closingDate.getTime()) continue;

      const days = calculateDaysBetween(prevDate, payDate); // ✅ Math.round
      const interest = roundToKopecks(balance * annualRate * days / 365); // ✅ Округляем проценты
      const principalPart = roundToKopecks(annuityPayment - interest); // ✅ Округляем тело
      
      balance = roundToKopecks(balance - principalPart); // ✅ Округляем остаток
      if (balance < 0) balance = 0;
      
      totalInterest += interest;
      
      rows.push({
        month: monthNumber++,
        date: payDate.toLocaleDateString('ru-RU'),
        payment: annuityPayment,
        interest,
        principal: principalPart,
        remainingBalance: balance,
      });
      prevDate = payDate;
    }

    // Финальный платёж — остаток + проценты
    if (balance > 0.01) {
      const daysFinal = calculateDaysBetween(prevDate, closingDate); // ✅ Math.round
      const intFinal = roundToKopecks(balance * annualRate * daysFinal / 365);
      const payFinal = roundToKopecks(balance + intFinal);
      
      rows.push({
        month: monthNumber,
        date: closingDate.toLocaleDateString('ru-RU'),
        payment: payFinal,
        interest: intFinal,
        principal: balance,
        remainingBalance: 0,
      });
    }
  }

  const totalPayments = roundToKopecks(rows.reduce((sum, row) => sum + row.payment, 0));
  const totalPrincipal = roundToKopecks(rows.reduce((sum, row) => sum + row.principal, 0));
  const monthlyRate = annualRate / 12;
  const calculatedAnnuity = isDiff
    ? (rows[0]?.payment || 0)
    : roundToKopecks(
        principal * (monthlyRate * Math.pow(1 + monthlyRate, months)) /
        (Math.pow(1 + monthlyRate, months) - 1)
      );

  return {
    monthlyPayment: calculatedAnnuity,
    totalPayments,
    totalInterest: roundToKopecks(totalInterest),
    totalPrincipal,
    schedule: rows,
  };
}

export function calculateTotalPayments(debt: Debt): number {
  if (
    debt.type !== 'bank_loan' ||
    !debt.interestRate ||
    !debt.termMonths ||
    !debt.startDate
  ) {
    return debt.currentAmount;
  }
  return calculatePaymentSchedule(debt).totalPayments;
}
